## 适用场景
按层处理二叉树节点；需要维护层级关系。
特征：看到"层序遍历"、"锯齿形遍历"、"每层的最值"、
"二叉树的右视图"、"完全二叉树的判定"、"二叉树的最大宽度"。

## 代码模板
```python
from collections import deque

# 标准层序遍历
def level_order(root):
    if not root:
        return []
    res = []
    q = deque([root])
    while q:
        level = []
        for _ in range(len(q)):  # 按层处理
            node = q.popleft()
            level.append(node.val)
            if node.left:
                q.append(node.left)
            if node.right:
                q.append(node.right)
        res.append(level)
    return res
```

```python
# 锯齿形层序遍历
def zigzag_level_order(root):
    if not root:
        return []
    res = []
    q = deque([root])
    reverse = False
    while q:
        level = []
        for _ in range(len(q)):
            node = q.popleft()
            level.append(node.val)
            if node.left:
                q.append(node.left)
            if node.right:
                q.append(node.right)
        if reverse:
            level.reverse()
        res.append(level)
        reverse = not reverse
    return res
```
